另一种方法

首先依然是$dp \cdot e \equiv 1 \mod (p-1)$，$d \cdot e = 1 + k \cdot (p-1)$我们利用费马小定理$a^{p-1} \equiv 1 \mod p$，构造$a^{dp \cdot e} \equiv a \mod p$，则$p | (a^{dp \cdot e}-a)$，和n求个gcd即可求出p，一般a取2就行

另外注意一点，直接算pow(a, dp * e)不行，没有模数，算出来的数太大了，而且耗时太长，我们取模数n即可，因为

$$ k \cdot p = a^{dp \cdot e}-a = ( a^{dp \cdot e} \mod n) + t \cdot n -a $$

把t*n移到左边，p依然整除pow(a, dp * e, n) - a，

最后，这个题的密文给的就是个数字，直接解就行

from random import randbytes

def menu():
    print(f"1. 加密消息")
    print(f"2. 校验明文是否正确")
    print(f"3. 退出")


class Key:
    def __init__(self):
        self.seed = randbytes(msg_len)

    def generate(self):
        self.seed = self.seed[offset:] + self.seed[:offset]
        return self.seed


def pad(msg):
    pad_len = msg_len - len(msg)
    return msg + pad_len * long_to_bytes(pad_len)


def encrypt(msg, key):
    cipher = xor(pad(msg), key)
    return b64encode(cipher)


def main():
    banner()
    key = Key()
    cur_key = key.generate()
    cipher1 = encrypt(plaintext, cur_key)
    print(f'[*] 首次密文(base64):{cipher1}\n')
    while True:
        try:
            menu()
            choice = int(input(f"[?] 请输入你的选择:"))
            if choice == 1:
                msg = input(f"[?] 请输入要加密的消息(长度小于等于{msg_len}): ").encode()
                if len(msg) > msg_len:
                    print(f"[!] 输入消息过长，最长为 {msg_len} 字节\n")
                    continue

                cur_key = key.generate()
                cipher = encrypt(msg, cur_key)

                print(f"[*] 你的消息已加密(密文): {cipher}\n")
                continue

            if choice == 2:
                msg = input(f"[?] 请输入待校验的明文: ").encode()

                    print(f"[*] 这是你的flag: {flag}\n")

def aes_encrypt(key, flag):
    aes = AES.new(key, AES.MODE_ECB)

def AMM(o, r, q):
    # o^(1/r) mod q
    start = time.time()
    print(
        '\n----------------------------------------------------------------------------------'
    )
    print('Start to run Adleman-Manders-Miller Root Extraction Method')
    print('Try to find one {:#x}th root of {} modulo {}'.format(r, o, q))
    g = GF(q)
    o = g(o)
    p = g(random.randint(1, q))
    while p ^ ((q - 1) // r) == 1:
        p = g(random.randint(1, q))
    print('[+] Find p:{}'.format(p))
    t = 0
    s = q - 1
    while s % r == 0:
        t += 1
        s = s // r
    print('[+] Find s:{}, t:{}'.format(s, t))
    k = 1
    while (k * s + 1) % r != 0:
        k += 1
    alp = (k * s + 1) // r
    print('[+] Find alp:{}'.format(alp))
    a = p ^ (r**(t - 1) * s)
    b = o ^ (r * alp - 1)
    c = p ^ s
    h = 1
    for i in range(1, t):
        d = b ^ (r ^ (t - 1 - i))
        if d == 1:
            j = 0
        else:

            j = -discrete_log(d, a)
            print('[+] Finish DLP...')

        h = h * c ^ j
        c = c ^ r
    result = o ^ alp * h
    end = time.time()
    print("Finished in {} seconds.".format(end - start))
    print('Find one solution: {}'.format(result))
    return result

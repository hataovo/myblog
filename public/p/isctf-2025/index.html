def check_equal(P, Q):
    Px, Py = P[0], P[1]
    Qx, Qy = Q[0], Q[1]
    if Px == Qx and Py == Qy:
        return True
    else:
        return False

# None表示无穷远点
def point_add(P, Q):
    if P is None:
        return Q
    if Q is None:
        return P
    x1, y1 = P[0], P[1]
    x2, y2 = Q[0], Q[1]
    if x1 == x2 and y1 != y2:
        return None

    if check_equal(P, Q):  # P=Q
        if y1 == 0:
            return None
        tmp = pow(2 * y1, -1, p)
        lamda = (3 * pow(x1, 2, p) + a) * tmp % p
        x3 = (pow(lamda, 2, p) - 2 * x1) % p
        y3 = (lamda * (x1 - x3) - y1) % p
        return (x3, y3)
    else:  # P!=Q
        tmp = pow(x2 - x1, -1, p)
        lamda = (y2 - y1) * tmp % p
        x3 = (pow(lamda, 2, p) - x1 - x2) % p
        y3 = (lamda * (x1 - x3) - y1) % p
        return (x3, y3)

# 方式1，从低向高
def beidian1(k, P):
    if k == 0 or P is None:
        return None

    result = None
    tmp = P
    while k:
        if k & 1:
            if result is None:
                result = tmp
            else:
                result = point_add(result, tmp)
        tmp = point_add(tmp, tmp)
        k >>= 1
    return result

# 方式2，从高向低
def beidian2(k, P):
    if k == 0 or P is None:
        return None

    k_bin = bin(k)[2:]
    tmp = None
    for i in range(len(k_bin)):
        tmp = point_add(tmp, tmp)
        if k_bin[i] == '1':
            tmp = point_add(tmp, P)
        else:
            pass
    return tmp

import math

def bsgs_ecc(G, Q, bound):
    # 求k，Q = k * G

    # 1. 计算步长 m = ceil(sqrt(bound))
    m = int(math.isqrt(bound)) + 1
    print(f"[BSGS] 步长 m = {m}")

    # 2. 预计算 baby steps: j × G for j = 0..m-1
    print(f"[BSGS] 预计算 baby steps (0 ≤ j < {m})...")
    baby_steps = {}

    # 0 × G = 无穷远点
    baby_steps[None] = 0

    # 1 × G
    current = G
    baby_steps[current] = 1

    # 计算 j × G for j = 2..m-1
    for j in range(2, m):
        current = point_add(current, G)
        if current is None:
            continue
        baby_steps[current] = j

    print(f"[BSGS] 预计算完成，存储了 {len(baby_steps)} 个点")

    # 3. 计算大步：m × G
    print(f"[BSGS] 计算大步：m × G...")
    mG = beidian1(m, G)

    if mG is None:
        print(f"[BSGS] 错误：m × G 是无穷远点")
        return None

    print(f"[BSGS] m × G = {hex(mG[0])[:20]}..., {hex(mG[1])[:20]}...")

    # 4. 计算 -mG（mG的逆元）
    mG_inv = (mG[0], (-mG[1]) % p)

    # 5. 搜索 giant steps: Q - i × (m × G)
    print(f"[BSGS] 开始搜索 giant steps (0 ≤ i < {m})...")

    # 初始化当前点为 Q
    current = Q

    for i in range(m):
        # 检查进度
        if i % (m // 10) == 0 and i > 0:
            print(f"[BSGS] 进度: {i}/{m} ({i/m*100:.1f}%)")

        # 检查当前点是否在baby steps中
        if current in baby_steps:
            j = baby_steps[current]
            k = (i * m + j) % bound

            # 验证结果
            test_point = beidian1(k, G)
            if test_point == Q:

                print(f"[BSGS] 找到解！k = {k}")
                print(f"[BSGS] 验证: {k} × G = Q")

                return k

        # 更新当前点：current = current - mG
        current = point_add(current, mG_inv)
        if current is None:
            # 如果遇到无穷远点，继续下一个i
            continue

    print(f"[BSGS] 在范围 0 ≤ k < {bound} 内未找到解")

    return None

def SmartAttack(P, Q, p):
    # Q = k * P
    E = P.curve()
    Eqp = EllipticCurve(Qp(p, 2),
                        [ZZ(t) + randint(0, p) * p for t in E.a_invariants()])

    P_Qps = Eqp.lift_x(ZZ(P.xy()[0]), all=True)
    for P_Qp in P_Qps:
        if GF(p)(P_Qp.xy()[1]) == P.xy()[1]:
            break

    Q_Qps = Eqp.lift_x(ZZ(Q.xy()[0]), all=True)
    for Q_Qp in Q_Qps:
        if GF(p)(Q_Qp.xy()[1]) == Q.xy()[1]:
            break

    p_times_P = p * P_Qp
    p_times_Q = p * Q_Qp

    x_P, y_P = p_times_P.xy()
    x_Q, y_Q = p_times_Q.xy()

    phi_P = -(x_P / y_P)
    phi_Q = -(x_Q / y_Q)
    k = phi_Q / phi_P
    return ZZ(k)
